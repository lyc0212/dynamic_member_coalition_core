class NestedJournal:
 def __init__(self,j):self.j=j;self.stack=[]
 @property
 def sequence(self):return self.j.sequence
 def append(self,e):
  e=deepcopy(e);status=e['status'];call=e['call_id']
  if status=='STARTED':e['parent_call_id']=self.stack[-1] if self.stack else None
  elif status in ('RETURNED','FAILED'):
   require(self.stack and self.stack[-1]==call,'CALL_STACK_MISMATCH');e['parent_call_id']=self.stack[-2] if len(self.stack)>1 else None
  else:e['parent_call_id']=self.stack[-1] if self.stack else None
  result=self.j.append(e)
  if status=='STARTED':self.stack.append(call)
  elif status in ('RETURNED','FAILED'):self.stack.pop()
  return result

def audit_calls(rows):
 stack=[];started={};finished={};rejected=[]
 for row in rows:
  e=row['event']
  if e.get('kind')!='WLS':continue
  cid=e['call_id'];s=e['status']
  if s=='STARTED':
   require(cid not in started and e['parent_call_id']==(stack[-1] if stack else None),'DUPLICATE_OR_PARENT_CALL');started[cid]=e;stack.append(cid)
  elif s in ('RETURNED','FAILED'):
   require(cid in started and cid not in finished and stack and stack[-1]==cid,'CALL_TERMINATION');require(e['category']==started[cid]['category'],'CALL_CATEGORY');finished[cid]=e;stack.pop()
  elif s=='REJECTED':require(cid not in started,'REJECTED_STARTED');rejected.append(e)
 require(not stack and set(started)==set(finished),'UNCLOSED_CALLS');require(not rejected and all(e['status']=='RETURNED' for e in finished.values()),'FAILED_OR_REJECTED_CALLS')
 return dict(status='PASS',started=len(started),returned=len(finished),counts={k:sum(e['category']==k for e in started.values()) for k in LIMITS})
